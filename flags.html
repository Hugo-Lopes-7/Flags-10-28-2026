countries = transpose(flag_data)

// Hidden reactive counter for drawing a new question
viewof next_btn = {
  const form = Inputs.button("Next Question", { value: 0 });
  form.classList.add("ojs-hidden-btn");
  return form;
}

// Choose a random country each time the page loads or "Next Question" is clicked
question = {
  next_btn;
  const idx = Math.floor(Math.random() * countries.length);
  const target = countries[idx];
  
  const options = [
    { file: target.file, name: target.name, correct: true },
    { file: target.similar[0], name: target.similar_names[0], correct: false },
    { file: target.similar[1], name: target.similar_names[1], correct: false },
    { file: target.similar[2], name: target.similar_names[2], correct: false }
  ];
  
  // Shuffle the 4 flag options
  for (let i = options.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [options[i], options[j]] = [options[j], options[i]];
  }
  
  return { target, options };
}

    const card = html`<button class="flag-option">
      <img src="flags/${opt.file}" alt="Flag option">
    </button>`;

    card.onclick = () => {
      if (opt.correct) {
        card.style.borderColor = "#22c55e";
        card.style.background = "rgba(22, 163, 74, 0.25)";
        feedback.style.color = "#4ade80";
        feedback.style.borderColor = "#22c55e";
        feedback.textContent = `✅ Correct! Here are some quick facts about ${opt.name}:`;
        factBox.style.display = "block";
      } else {
        card.style.borderColor = "#ef4444";
        card.style.background = "rgba(220, 38, 38, 0.25)";
        feedback.style.color = "#f87171";
        feedback.style.borderColor = "#ef4444";
        feedback.textContent = `❌ Wrong! That is the flag of ${opt.name}. Try another flag!`;
      }
    };

    grid.appendChild(card);
  });

  nextTrigger.onclick = () => {
    const btn = document.querySelector(".ojs-hidden-btn button");
    if (btn) btn.click();
  };
